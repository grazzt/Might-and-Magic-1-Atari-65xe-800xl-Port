# Original v2: boot side A, swap to B, take the six supplied characters out of
# the Inn of Sorpigal, walk to the town gate and swap to side C outdoors.
# The route is the one in c64/images/Might_and_Magic_1_Achtung.txt.
until $41ab          # title keyboard poll
mark title
key 5f               # left arrow: the game's Escape
wait $1000000        # this prompt polls the keyboard itself, not through $0E3B
mark insert-b
attach B
wait $100000         # let the drive notice the new disk
key 0d
ready
mark menu
press 31             # town 1, Sorpigal
ready
mark inn
press 01             # CTRL-A adds a character
until $78a2          # the inn is about to test the KERNAL CTRL flag
poke 028e 04
press 02             # CTRL-B adds a character
until $78a2          # the inn is about to test the KERNAL CTRL flag
poke 028e 04
press 03             # CTRL-C adds a character
until $78a2          # the inn is about to test the KERNAL CTRL flag
poke 028e 04
press 04             # CTRL-D adds a character
until $78a2          # the inn is about to test the KERNAL CTRL flag
poke 028e 04
press 05             # CTRL-E adds a character
until $78a2          # the inn is about to test the KERNAL CTRL flag
poke 028e 04
press 06             # CTRL-F adds a character
until $78a2          # the inn is about to test the KERNAL CTRL flag
poke 028e 04
ready
mark party
press 58             # X: leave the inn
ready
mark town
press 49             # I
press 49             # I
press 49             # I
press 49             # I
press 4c             # L
press 49             # I
press 49             # I
press 49             # I
press 49             # I
press 4c             # L
press 49             # I
press 49             # I
press 49             # I
press 49             # I
press 49             # I
press 49             # I
press 49             # I
ready
mark gate
press 59             # Y: leave town
ready
mark insert-c
attach C
wait $100000         # let the drive notice the new disk
ready
poke 002d 00 00      # the wait spun the key-wait loop: reset its two counters
press 0d
ready
mark outdoors
